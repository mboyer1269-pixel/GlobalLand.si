# Local CSV structure check

Run: python3 products/invoice-preflight/csv_structure_check.py products/invoice-preflight/sample.csv

Checks comma-separated UTF-8 CSV syntax, blank or duplicate headers and inconsistent record widths. Runs locally using the Python standard library. Multiline quoted fields and UTF-8 BOM are supported.

This is a generic structural check. It does not validate invoice amounts, tax, customer mappings, required platform fields or successful import into Manager or any accounting platform. Passing does not establish import compatibility. No payment or revenue has been verified.

=== invoice-preflight/csv_structure_check.py ===
import csv, io, json, sys

def check(text):
    issues=[]
    try:
        rows=list(csv.reader(io.StringIO(text.lstrip("\ufeff")),strict=True))
    except csv.Error as exc:
        return {"issues":[{"kind":"invalid_csv","detail":str(exc)}],"data_rows":0}
    if not rows:
        return {"issues":[{"kind":"empty_file"}],"data_rows":0}
    headers=rows[0]
    if any(not h.strip() for h in headers): issues.append({"kind":"blank_header"})
    normalized=[h.strip().casefold() for h in headers]
    if len(set(normalized))!=len(normalized): issues.append({"kind":"duplicate_header"})
    for line,row in enumerate(rows[1:],2):
        if len(row)!=len(headers): issues.append({"kind":"column_count","record":line,"expected":len(headers),"actual":len(row)})
    return {"issues":issues,"data_rows":len(rows)-1,"columns":len(headers)}

if __name__=="__main__":
    with open(sys.argv[1],encoding="utf-8",newline="") as source: result=check(source.read())
    print(json.dumps(result,indent=2))
    sys.exit(1 if result["issues"] else 0)
